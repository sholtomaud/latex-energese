% The aggregated-economy model authored in TeX rather than JSON. This is the
% same system as examples/aggregated-economy.json and renders identically --
% `make parity` asserts it. It is the complex counterpart to the simple
% source-store and simple_system pairs: manual coordinates, a system window, a
% relocated heat sink, interaction symbols, and several money-feedback arcs.
\documentclass{standalone}
\usepackage{energese}
\begin{document}
\begin{energese}
    \esmeta{heat_sink_label={11,297.6}, ground_x=7.44, heat_sink_y=-3.72}
    \esboundary{x_min=1.3, x_max=14.01, y_min=-3.47, y_max=3.72,
                label={$10^{15}$ Calories per Year}}

    \esnode{sun}{source}       {Tr=1,     x=0, y=0,  magnitude=1.5, label={Sun}}
    \esnode{fuels}{source}     {Tr=1000,  x=8.18, y=4.96,  label={Fuels,\\Minerals}}
    \esnode{env_systems}{producer}{Tr=10, x=2.6, y=0.62,  label={Environmental\\Systems}}
    \esnode{agri}{producer}    {Tr=10,    x=5.21, y=-1.86, label={Agriculture}}
    % Interaction symbols: where separate inflows must combine before they can
    % be used. Odum draws these as the notched shapes abutting each consumer.
    \esnode{ix_industry}{interaction}{Tr=1000,  x=6.45, y=1.24}
    \esnode{industry}{consumer}{Tr=2000,  x=8.18, y=1.24,  magnitude=1.5, label={Industry}}
    \esnode{consumer}{consumer}{Tr=10000, x=12.15, y=-0.62, label={Consumer\\Human Service}}
    % Exchanges: goods one way, money the other, coupled at a diamond.
    \esnode{ex_ic}{transaction}{seller=industry,
                                buyer=consumer, money_label={\$ $1.1 \times 10^{12}$}}
    \esnode{ex_ai}{transaction}{seller=agri,
                                buyer=ix_industry, money_label={\$}}
    \esnode{ex_ac}{transaction}{seller=agri,
                                buyer=consumer, money_label={\$}}
    \esnode{human_service_label}{text}{x=12.15, y=3.41,   label={Human Service}}
    % Energy reaching the source but not used by the system: Odum draws it as a
    % branch that leaves without driving anything.
    \esnode{unused}{text}{x=0.74, y=-2.11}

    \esflow{sun}{env_systems}{energy}   {label={6000}, options={bend left=20}}
    \esflow{sun}{agri}{energy}          {label={5280}, options={bend right=20}}
    \esflow{sun}{unused}{energy}        {options={bend left=25}}
    \esflow{fuels}{ix_industry}{energy} {label={17.6}, from_anchor=south,
                                         to_anchor=feedback_in}
    \esflow{env_systems}{ix_industry}{energy}{options={bend left=8}}
    \esflow{ix_industry}{industry}{energy}
\end{energese}
\end{document}
